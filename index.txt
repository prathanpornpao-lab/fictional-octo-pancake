        // Levenshtein Distance for Similarity
        function stringSimilarity(s1, s2) {
            s1 = s1.toLowerCase().replace(/[^a-z]/g, '');
            s2 = s2.toLowerCase().replace(/[^a-z]/g, '');
            
            // Direct match or contains
            if(s1 === s2 || s1.includes(s2) || s2.includes(s1)) return 100;

            if (s1.length === 0) return 0;
            if (s2.length === 0) return 0;

            const matrix = [];
            for (let i = 0; i <= s2.length; i++) {
                matrix[i] = [i];
            }
            for (let j = 0; j <= s1.length; j++) {
                matrix[0][j] = j;
            }

            for (let i = 1; i <= s2.length; i++) {
                for (let j = 1; j <= s1.length; j++) {
                    if (s2.charAt(i - 1) === s1.charAt(j - 1)) {
                        matrix[i][j] = matrix[i - 1][j - 1];
                    } else {
                        matrix[i][j] = Math.min(
                            matrix[i - 1][j - 1] + 1, // substitution
                            Math.min(matrix[i][j - 1] + 1, // insertion
                            matrix[i - 1][j] + 1) // deletion
                        );
                    }
                }
            }
            const distance = matrix[s2.length][s1.length];
            const maxLength = Math.max(s1.length, s2.length);
            return Math.round(((maxLength - distance) / maxLength) * 100);
        }

        // Helper to get pixel coordinates from normalized landmarks
        function getCoords(landmark, w, h) {
            return { x: landmark.x * w, y: landmark.y * h };

        function drawAREffects(ctx, landmarks, w, h) {
            if (gameState.arEffect === 'none') return;

            // Find key points
            const topHead = getCoords(landmarks[10], w, h); // Top center forehead
            const leftCheek = getCoords(landmarks[234], w, h);
            const rightCheek = getCoords(landmarks[454], w, h);
            const faceWidth = Math.abs(rightCheek.x - leftCheek.x);

            if (gameState.arEffect === 'pass') {
                // Draw Crown
                const crownWidth = faceWidth * 0.9;
                const crownHeight = crownWidth * 0.6;
                const cx = topHead.x;
                const cy = topHead.y - (crownHeight * 0.4); // Offset up

                ctx.fillStyle = '#FFD700'; // Gold
                ctx.strokeStyle = '#DAA520';
                ctx.lineWidth = 3;
                ctx.lineJoin = 'round';

                ctx.beginPath();
                ctx.moveTo(cx - crownWidth/2, cy); // Bottom left
                ctx.lineTo(cx + crownWidth/2, cy); // Bottom right
                ctx.lineTo(cx + crownWidth/2 + 10, cy - crownHeight*0.6); // Right edge up
                ctx.lineTo(cx + crownWidth/4, cy - crownHeight*0.3); // Inner right
                ctx.lineTo(cx, cy - crownHeight); // Top center peak
                ctx.lineTo(cx - crownWidth/4, cy - crownHeight*0.3); // Inner left
                ctx.lineTo(cx - crownWidth/2 - 10, cy - crownHeight*0.6); // Left edge up
                ctx.closePath();
                
                ctx.fill();
                ctx.stroke();

                // Jewels
                ctx.fillStyle = '#FF3366'; // Red jewel
                ctx.beginPath(); ctx.arc(cx, cy - crownHeight*0.4, crownWidth*0.08, 0, Math.PI*2); ctx.fill();
                ctx.fillStyle = '#33CCFF'; // Blue jewel
                ctx.beginPath(); ctx.arc(cx - crownWidth*0.3, cy - crownHeight*0.25, crownWidth*0.05, 0, Math.PI*2); ctx.fill();
                ctx.beginPath(); ctx.arc(cx + crownWidth*0.3, cy - crownHeight*0.25, crownWidth*0.05, 0, Math.PI*2); ctx.fill();

            } else if (gameState.arEffect === 'fail') {
                // Draw Devil Horns
                const leftTemple = getCoords(landmarks[54], w, h); // roughly left top
                const rightTemple = getCoords(landmarks[284], w, h); // roughly right top
                const hornBase = faceWidth * 0.15;
                const hornHeight = faceWidth * 0.35;

                ctx.fillStyle = '#DC2626'; // Red

                // Right Horn (viewer perspective, canvas is mirrored)
                ctx.beginPath();
                ctx.moveTo(rightTemple.x - hornBase/2, rightTemple.y);
                ctx.lineTo(rightTemple.x + hornBase/2, rightTemple.y);
                ctx.quadraticCurveTo(rightTemple.x + hornBase, rightTemple.y - hornHeight/2, rightTemple.x + hornBase*1.5, rightTemple.y - hornHeight);
                ctx.quadraticCurveTo(rightTemple.x, rightTemple.y - hornHeight/2, rightTemple.x - hornBase/2, rightTemple.y);
                ctx.fill();

                // Left Horn
                ctx.beginPath();
                ctx.moveTo(leftTemple.x - hornBase/2, leftTemple.y);
                ctx.lineTo(leftTemple.x + hornBase/2, leftTemple.y);
                ctx.quadraticCurveTo(leftTemple.x - hornBase, leftTemple.y - hornHeight/2, leftTemple.x - hornBase*1.5, leftTemple.y - hornHeight);
                ctx.quadraticCurveTo(leftTemple.x, leftTemple.y - hornHeight/2, leftTemple.x + hornBase/2, leftTemple.y);
                ctx.fill();
            }
        }

        function drawVignette(ctx, w, h) {

            ctx.fillRect(0, 0, w, h);
        }

            gameState.status = 'counting';
            let count = 3;
            
            DOM.statusBadge.innerText = "เตรียมตัว...";
            DOM.statusBadge.className = "mb-4 px-4 py-1 rounded-full bg-blue-100 text-blue-600 font-medium transition-opacity opacity-100";
            
            const countInterval = setInterval(() => {
                if (count > 0) {
                    DOM.statusBadge.innerText = `เริ่มใน ${count}...`;
                    AudioEngine.tick();
                    DOM.englishWord.style.transform = `scale(${1 + (4-count)*0.05})`; // Slight bounce
                    count--;
                } else {
                    clearInterval(countInterval);
                    startListening();
                }
            }, 1000);
        }

        function startListening() {
            gameState.status = 'listening';
            AudioEngine.go();
            
            DOM.statusBadge.innerText = "พูดเลย!";
            DOM.statusBadge.className = "mb-4 px-4 py-1 rounded-full bg-green-100 text-green-600 font-bold transition-opacity opacity-100 animate-pulse";
            
            DOM.micIndicator.classList.add('mic-active');
            DOM.micIndicator.querySelector('svg').classList.replace('text-gray-400', 'text-green-500');

            try {
                recognition.start();
                
                // 5 Second Timeout
                speechTimeout = setTimeout(() => {
                    if (gameState.status === 'listening') {
                        recognition.stop();
                        handleNoSpeech();
                    }
                }, 5000);
            } catch (e) {
                console.error(e);
                handleNoSpeech();
            }
        }

        function resetMeter() {
            DOM.accuracyBar.style.width = '0%';
            DOM.accuracyBar.className = "h-full w-0 transition-all duration-500 ease-out bg-gray-400";
            DOM.accuracyText.innerText = "0%";
            DOM.resultMessage.innerText = "";
            DOM.micIndicator.classList.remove('mic-active');
            DOM.micIndicator.querySelector('svg').classList.replace('text-green-500', 'text-gray-400');
        }

        
        function handleSpeechResult(transcript) {
            if (gameState.status !== 'listening') return;
            gameState.status = 'showing_result';
            
            const targetWord = vocabularyList[gameState.currentIndex].en;
            const similarity = stringSimilarity(targetWord, transcript);
            
            console.log(`Target: ${targetWord}, Spoken: ${transcript}, Score: ${similarity}%`);
            
            if (similarity >= 80) {
                showResult('pass', similarity, `พูดว่า: "${transcript}"`);
            } else {
                showResult('fail', similarity, `พูดว่า: "${transcript}"`);
            }
        }

        function handleNoSpeech() {
            if (gameState.status !== 'listening') return;
            gameState.status = 'showing_result';
            
            AudioEngine.warning();
            
            DOM.statusBadge.innerText = "ไม่ได้ยินเสียง";
            DOM.statusBadge.className = "mb-4 px-4 py-1 rounded-full bg-yellow-100 text-yellow-600 font-medium transition-opacity opacity-100";
            
            resetMeter();
            DOM.accuracyBar.style.width = '100%';
            DOM.accuracyBar.className = "h-full transition-all duration-500 ease-out bg-yellow-400";
            DOM.accuracyText.innerText = "-";
            DOM.resultMessage.innerText = "ลองพูดดังๆ อีกครั้งนะ";
            DOM.resultMessage.className = "text-center mt-3 font-medium h-6 text-sm text-yellow-600";

            // Delay and restart same question
            setTimeout(() => {
                showQuestion();
            }, 3000);
        }

        function showResult(status, percentage, msgText) {
            DOM.micIndicator.classList.remove('mic-active');
            DOM.micIndicator.querySelector('svg').classList.replace('text-green-500', 'text-gray-400');
            
            DOM.accuracyBar.style.width = `${percentage}%`;
            DOM.accuracyText.innerText = `${percentage}%`;
            DOM.resultMessage.innerText = msgText;

            if (status === 'pass') {
                gameState.arEffect = 'pass';
                gameState.score += 1;
                updateScoreUI();
                AudioEngine.correct();
                
                DOM.statusBadge.innerText = "ผ่าน!";
                DOM.statusBadge.className = "mb-4 px-4 py-1 rounded-full bg-green-100 text-green-600 font-bold opacity-100";
                
                DOM.accuracyBar.className = "h-full transition-all duration-500 ease-out bg-green-500";
                DOM.resultMessage.className = "text-center mt-3 font-medium h-6 text-sm text-green-600";
                
                // Fire confetti on the left side
                confetti({
                    particleCount: 100,
                    spread: 70,
                    origin: { x: 0.25, y: 0.6 } // Approx left panel center
                });

            } else {
                gameState.arEffect = 'fail';
                AudioEngine.wrong();
                
                DOM.statusBadge.innerText = "เกือบแล้ว!";
                DOM.statusBadge.className = "mb-4 px-4 py-1 rounded-full bg-red-100 text-red-600 font-bold opacity-100";
                
                DOM.accuracyBar.className = "h-full transition-all duration-500 ease-out bg-red-500";
                DOM.resultMessage.className = "text-center mt-3 font-medium h-6 text-sm text-red-600";
            }

            // Wait 3 seconds before next
            setTimeout(() => {
                nextQuestion();
            }, 3000);
        }

        function updateScoreUI() {
            DOM.currentScore.innerText = gameState.score;
        }

        function nextQuestion() {
            gameState.currentIndex++;
            if (gameState.currentIndex >= vocabularyList.length) {
                endGame();
            } else {
                showQuestion();
            }
        }

        function endGame() {
            gameState.arEffect = 'none';
            DOM.endScreen.classList.remove('hidden');
            DOM.finalScore.innerText = gameState.score;
            
            if (gameState.score >= 8) {
                DOM.endTitle.innerText = "สุดยอดไปเลย!";
                DOM.endTitle.className = "text-5xl font-semibold mb-2 text-green-500";
                AudioEngine.fanfare();
                // Big confetti
                var duration = 3000;
                var end = Date.now() + duration;
                (function frame() {
                    confetti({
                        particleCount: 5,
                        angle: 60,
                        spread: 55,
                        origin: { x: 0 },
                        colors: ['#26ccff', '#a25afd', '#ff5e7e', '#88ff5a', '#fcff42', '#ffa62d', '#ff36ff']
                    });
                    confetti({
                        particleCount: 5,
                        angle: 120,
                        spread: 55,
                        origin: { x: 1 },
                        colors: ['#26ccff', '#a25afd', '#ff5e7e', '#88ff5a', '#fcff42', '#ffa62d', '#ff36ff']
                    });
                    if (Date.now() < end) {
                        requestAnimationFrame(frame);
                    }
                }());
            } else if (gameState.score >= 5) {
                DOM.endTitle.innerText = "เก่งมาก!";
                DOM.endTitle.className = "text-5xl font-semibold mb-2 text-orange-500";
                AudioEngine.correct();
            } else {
                DOM.endTitle.innerText = "พยายามอีกนิดนะ!";
                DOM.endTitle.className = "text-5xl font-semibold mb-2 text-gray-600";
                AudioEngine.gameover();
            }
        }

        // Start camera feed on load
        camera.start().catch(err => {
            console.error("Camera error:", err);
            DOM.startBtn.innerText = "ไม่พบกล้อง / ถูกปฏิเสธสิทธิ";
            DOM.cameraLoading.innerHTML = "<p class='text-red-400'>ไม่สามารถเข้าถึงกล้องได้</p>";
        });
